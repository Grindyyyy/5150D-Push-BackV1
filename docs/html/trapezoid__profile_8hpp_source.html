66 static TrapezoidProfile from_constraints(
67 double total_distance,
68 double max_velocity,
69 double max_acceleration
70 ) {
71 bool invert = false;
72
73 if (total_distance < 0) {
74 invert = true;
75
76 }
77
78 auto abs_max_velocity = std::abs(max_velocity);
79 auto abs_max_acceleration = std::abs(max_acceleration);
80 auto abs_total_distance = std::abs(total_distance);
81
82 // get the time it takes to accelerate to max velocity
83 auto accel_time = abs_max_velocity / abs_max_acceleration;
84
85 // the deceleration time is the same
86 auto decel_time = accel_time;
87
88 // get the acceleration distance via x = 1/2at^2
89 auto accel_distance = (abs_max_acceleration * std::pow((accel_time),2) / 2);
90
91 // the deceleration distance is the same
92 auto decel_distance = accel_distance;
93
94 // gets coast distance by total - (accel + decel)
95 auto coast_distance = abs_total_distance - accel_distance - decel_distance;
96
97 // if the coast distance is less than zero, compute the maximum acceleration we can reach in the time given
98 if (coast_distance < 0) {
99 // find the accel time via the equation sqrt(2x/a) = t
100 accel_time = std::sqrt(abs_total_distance / abs_max_acceleration);
101
102 // decel time will be the same as accel time
103 decel_time = accel_time;
104
105 // no coast distance
106 coast_distance = 0;
107 }
108
109 // compute the amount of time we want to coast for
110 auto coast_time = coast_distance / abs_max_velocity;
111
112 // total time is all segments added together
113 auto total_time = accel_time + decel_time + coast_time;
114
115 auto accel_cutoff = accel_time;
116 auto coast_cutoff = accel_time + coast_time;
117 auto decel_cutoff = accel_time + coast_time + decel_time;
118
119 return TrapezoidProfile(
120 abs_total_distance,
121 abs_max_velocity,
122 abs_max_acceleration,
123 total_time,
124 accel_cutoff,
125 coast_cutoff,
126 decel_cutoff,
127 invert
128 );
129 }
137 TrapezoidProfileStage stage(double elapsed_time) const {

155 ProfileSetpoint calculate(double elapsed_time) const {
156 // integration constant for the coasting segment
157 double const_1 =
158 -m_max_velocity * m_accel_cutoff
159 + (m_max_acceleration / 2) * std::pow(m_accel_cutoff,2);
160
161 // integration constant for the deceleration segment
162 double const_2 =
163 -m_max_acceleration * m_total_time * m_coast_cutoff
164 + (m_max_acceleration / 2) * std::pow(m_coast_cutoff,2)
165 + m_max_velocity * m_coast_cutoff + const_1;
166
167 ProfileSetpoint setpoint = ProfileSetpoint(0, 0, 0);
168
169 switch (this->stage(elapsed_time)) {
170 case TrapezoidProfileStage::Accelerating:
171 setpoint = ProfileSetpoint(
172 (m_max_acceleration / 2) * std::pow(elapsed_time,2),
173 m_max_acceleration * elapsed_time,
174 m_max_acceleration
175 );
176 break;
177 case TrapezoidProfileStage::Coasting:
178 setpoint = ProfileSetpoint(
179 m_max_velocity * elapsed_time + const_1,
180 m_max_velocity,
181 m_max_acceleration - m_max_acceleration // long ass way to say 0 bc complier complains
182 );
183 break;
184 case TrapezoidProfileStage::Decelerating:
185 setpoint = ProfileSetpoint(
186 m_max_acceleration * m_total_time * elapsed_time
187 - (m_max_acceleration / 2) * std::pow(elapsed_time,2)
188 + const_2,
189 m_max_acceleration * (m_total_time - elapsed_time),
190 -m_max_acceleration
191 );
192 break;
193 case TrapezoidProfileStage::Done:
194 setpoint = ProfileSetpoint(m_total_distance, 0, 0);
195 break;
196 default:
197 // to stop the compiler from complaining
198 setpoint = ProfileSetpoint(m_total_distance, 0, 0);
199 }
200
201 return setpoint.direct(m_invert);
202 }
